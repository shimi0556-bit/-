איך זה עובד

      
במקום להזיז קודקודים, מתארים את המודל כנוסחה: כל צורה יודעת להגיד כמה רחוקה כל נקודה במרחב מהמשטח שלה. מחברים, ממיסים וחורטים צורות, והמחשב מצייר את התוצאה ישירות מהנוסחה.

      
צורות בסיס

      
sphere(0.5)                 // כדור
box(1, 0.6, 0.4, 0.05)      // קופסה עם פינות מעוגלות
capsule([0,0,0], [0,1,0], 0.2, 0.1)  // גפה מחודדת
cylinder(0.3, 1) · cone(0.4, 0.1, 1) · torus(0.5, 0.1)
tube([[0,0,0,.1],[.3,.5,0,.07],[0,1,0,.03]])  // זנב/זרוע חלקה

      
חיבור, המסה וחריטה

      
a.add(b)          // חיבור רגיל
a.add(b, 0.1)     // המסה חלקה ברדיוס 0.1
a.cut(b, 0.02)    // חריטה עם קצה מעוגל
a.intersect(b)    // רק החלק המשותף

      
מיקום ושכפול

      
.move(x, y, z) · .rotate(0, 45, 0) · .scale(1.2)
.mirror('x')     // צד אחד, והשני נוצר בסימטריה
.ring(8)         // 8 עותקים מסביב לציר האנכי
.grid([0.3,0,0], [5,1,1])  // שורה של 5

      
צבע וחומר

      
.color('#c7372f', 'glossy')   // matte, satin, plastic, metal, chrome, glow…
.paint(sphere(0.1).move(0,1,0.4), '#fff')  // צביעת אזור
.colorNoise('#6b4f2a', 8)     // כתמים טבעיים
.displace(0.02, 10)           // מרקם אורגני

      
צורות מסובבות ומשוכות

      
polygon([[0,0],[.4,0],[.3,.5],[.45,1],[0,1]], {smooth: 8}).revolve()  // אגרטל
star(5, 0.5, 0.22).extrude(0.1, 0.02)   // כוכב תלת־ממדי

      
הנפשה

      
const tail = tube([...]).bone('tail', [0, 0.3, -0.5])  // חלק זז + נקודת ציר
animate({ seconds: 4 }, {
  tail: (t) => ({ rotate: [0, 35 * wave(t, 8), 0] }),   // כשכוש
  head: (t) => ({ rotate: [0, 0, 10 * wave(t, 1)] }),   // הטיית ראש
})

      
סליידרים

      
const ears = param('ears', 0.3, 0.1, 0.6)   // יוצר סליידר חי

      
הסוד לאיכות: קודם מסות גדולות, אחר כך צורות משניות, ורק בסוף פרטים קטנים. המסו חלקים אורגניים עם רדיוס של 10–30% מגודל החלק הקטן.
